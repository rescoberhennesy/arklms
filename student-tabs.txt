'use client';

import { useState } from 'react';
import Link from 'next/link';
import { useRouter } from 'next/navigation';
import {
  ArrowLeft,
  Play,
  RotateCw,
  Trophy,
  Calendar,
  Hash,
  FileText,
  BookOpen,
  Loader2,
  Trash2,
} from 'lucide-react';
import type { PracticeSessionDetail } from '@/lib/types/practice';
import {
  PRACTICE_DIFFICULTY_LABELS,
  PRACTICE_QUESTION_KIND_LABELS,
} from '@/lib/types/practice';
import {
  startPracticeAttempt,
  deletePracticeSession,
} from '@/lib/actions/practice';

interface Props {
  detail: PracticeSessionDetail;
}

export default function PracticeSessionDetailView({ detail }: Props) {
  const router = useRouter();
  const { session, questions, attempts } = detail;
  const [starting, setStarting] = useState(false);
  const [deleting, setDeleting] = useState(false);

  const completedAttempts = attempts.filter((a) => a.submittedAt !== null);
  const inProgressAttempt = attempts.find((a) => a.submittedAt === null) ?? null;

  const totalPoints = questions.reduce((sum, q) => sum + q.points, 0);
  const bestAttempt = completedAttempts.reduce<typeof completedAttempts[number] | null>(
    (best, a) => {
      if (a.score === null) return best;
      if (!best || (best.score ?? 0) < a.score) return a;
      return best;
    },
    null,
  );

  // Tally question kinds for the "what's in this session" pill row
  const kindCounts = questions.reduce<Record<string, number>>((acc, q) => {
    acc[q.questionKind] = (acc[q.questionKind] ?? 0) + 1;
    return acc;
  }, {});

  async function handleStart() {
    setStarting(true);
    try {
      const { attempt } = await startPracticeAttempt(session.id);
      router.push(`/student/practice/${session.id}/attempt/${attempt.id}`);
    } catch (err) {
      alert(err instanceof Error ? err.message : 'Failed to start attempt');
      setStarting(false);
    }
  }

  async function handleResume() {
    if (!inProgressAttempt) return;
    router.push(`/student/practice/${session.id}/attempt/${inProgressAttempt.id}`);
  }

  async function handleDelete() {
    if (!confirm('Delete this practice session and all attempts? This cannot be undone.')) return;
    setDeleting(true);
    try {
      await deletePracticeSession(session.id);
      router.push('/student/practice');
    } catch (err) {
      alert(err instanceof Error ? err.message : 'Failed to delete');
      setDeleting(false);
    }
  }

  return (
    <div className="mx-auto max-w-3xl space-y-6 p-4 sm:p-6">
      <Link
        href="/student/practice"
        className="inline-flex items-center gap-1 text-sm text-gray-500 hover:text-gray-900"
      >
        <ArrowLeft className="h-4 w-4" />
        Back to AI Practice
      </Link>

      {/* Header card */}
      <div className="rounded-xl border border-gray-200 bg-white p-5 shadow-sm">
        <div className="flex items-start justify-between gap-3">
          <div className="min-w-0 flex-1">
            <h1 className="text-xl font-semibold text-gray-900 sm:text-2xl">
              {session.title}
            </h1>
            <div className="mt-2 flex flex-wrap items-center gap-x-3 gap-y-1.5 text-xs text-gray-500">
              <span className="inline-flex items-center gap-1 rounded-full bg-indigo-50 px-2 py-0.5 font-semibold text-indigo-700">
                {PRACTICE_DIFFICULTY_LABELS[session.difficulty]}
              </span>
              <span className="inline-flex items-center gap-1">
                <Hash className="h-3 w-3" />
                {questions.length} questions · {totalPoints} pts
              </span>
              <span className="inline-flex items-center gap-1">
                <Calendar className="h-3 w-3" />
                Created {new Date(session.createdAt).toLocaleDateString()}
              </span>
              <span className="inline-flex items-center gap-1">
                {session.sourceKind === 'lesson' ? (
                  <>
                    <BookOpen className="h-3 w-3" />
                    From a lesson
                  </>
                ) : (
                  <>
                    <FileText className="h-3 w-3" />
                    From upload{session.sourceFileNames.length > 0 ? `: ${session.sourceFileNames.join(', ')}` : ''}
                  </>
                )}
              </span>
            </div>
          </div>
          <button
            type="button"
            onClick={handleDelete}
            disabled={deleting}
            aria-label="Delete session"
            className="rounded-md p-2 text-gray-400 hover:bg-red-50 hover:text-red-600 disabled:opacity-50"
          >
            <Trash2 className="h-4 w-4" />
          </button>
        </div>

        {session.sourceNote && (
          <p className="mt-3 rounded-md border border-amber-200 bg-amber-50 px-3 py-2 text-xs text-amber-900">
            <span className="font-semibold">Note from AI:</span> {session.sourceNote}
          </p>
        )}

        <div className="mt-4 flex flex-wrap items-center gap-1.5">
          {Object.entries(kindCounts).map(([kind, n]) => (
            <span
              key={kind}
              className="rounded-full bg-gray-100 px-2 py-0.5 text-[11px] font-medium text-gray-700"
            >
              {n}× {PRACTICE_QUESTION_KIND_LABELS[kind as keyof typeof PRACTICE_QUESTION_KIND_LABELS]}
            </span>
          ))}
        </div>

        {/* Primary action */}
        <div className="mt-5 flex flex-col gap-2 sm:flex-row">
          {inProgressAttempt ? (
            <button
              type="button"
              onClick={handleResume}
              className="inline-flex flex-1 items-center justify-center gap-2 rounded-lg bg-red-600 px-4 py-3 text-sm font-semibold text-white shadow-sm hover:bg-red-700"
            >
              <Play className="h-4 w-4" />
              Resume attempt
            </button>
          ) : (
            <button
              type="button"
              onClick={handleStart}
              disabled={starting || questions.length === 0}
              className="inline-flex flex-1 items-center justify-center gap-2 rounded-lg bg-red-600 px-4 py-3 text-sm font-semibold text-white shadow-sm hover:bg-red-700 disabled:cursor-not-allowed disabled:opacity-60"
            >
              {starting ? (
                <Loader2 className="h-4 w-4 animate-spin" />
              ) : completedAttempts.length === 0 ? (
                <Play className="h-4 w-4" />
              ) : (
                <RotateCw className="h-4 w-4" />
              )}
              {completedAttempts.length === 0 ? 'Start practice' : 'Retake'}
            </button>
          )}
        </div>
      </div>

      {/* Past attempts */}
      <section>
        <h2 className="mb-3 text-sm font-semibold text-gray-700">
          {completedAttempts.length === 0
            ? 'No attempts yet'
            : `Past attempts (${completedAttempts.length})`}
        </h2>

        {completedAttempts.length === 0 ? (
          <div className="rounded-xl border border-dashed border-gray-300 bg-white p-8 text-center">
            <Trophy className="mx-auto h-8 w-8 text-gray-300" />
            <p className="mt-2 text-sm italic text-gray-500">
              Take your first attempt to see your results here.
            </p>
          </div>
        ) : (
          <ul className="space-y-2">
            {completedAttempts.map((a, idx) => {
              const pct =
                a.score !== null && a.maxScore && a.maxScore > 0
                  ? Math.round((a.score / a.maxScore) * 100)
                  : 0;
              const isBest = bestAttempt?.id === a.id;
              return (
                <li
                  key={a.id}
                  className="rounded-lg border border-gray-200 bg-white p-3 shadow-sm"
                >
                  <div className="flex items-center justify-between gap-3">
                    <div className="min-w-0 flex-1">
                      <div className="flex items-center gap-2">
                        <span className="text-sm font-semibold text-gray-900">
                          Attempt #{completedAttempts.length - idx}
                        </span>
                        {isBest && (
                          <span className="inline-flex items-center gap-1 rounded-full bg-emerald-50 px-2 py-0.5 text-[10px] font-semibold text-emerald-700">
                            <Trophy className="h-3 w-3" />
                            Best
                          </span>
                        )}
                      </div>
                      <p className="mt-0.5 text-[11px] text-gray-500">
                        {a.submittedAt && new Date(a.submittedAt).toLocaleString()}
                      </p>
                    </div>
                    <div className="text-right">
                      <div
                        className={`text-base font-bold ${

                        {pct}%

                      <div className="text-[11px] text-gray-500">
                        {a.score}/{a.maxScore}

                  </div>

    </div>
  );

  prelim: 'text-blue-700',
  midterm: 'text-purple-700',

  prelim: 'bg-blue-500',
  midterm: 'bg-purple-500',

function CompletionRing({ pct }: { pct: number }) {
  const r = 34;
  const c = 2 * Math.PI * r;
  const filled = (Math.max(0, Math.min(100, pct)) / 100) * c;
  return (
    <svg role="img" aria-label={`${pct} percent submitted`} width="84" height="84" viewBox="0 0 84 84">
      <circle cx="42" cy="42" r={r} fill="none" stroke="#f4f4f5" strokeWidth="10" />
      {pct > 0 && (
        <circle
          cx="42"
          cy="42"
          r={r}
          fill="none"
          stroke="#16a34a"
          strokeWidth="10"
          strokeLinecap="round"
          strokeDasharray={`${filled} ${c}`}
          transform="rotate(-90 42 42)"
        />
      )}
      <text x="42" y="48" textAnchor="middle" fontSize="18" fontWeight="800" fill="#18181b">
        {pct}%
      </text>
    </svg>
  );
}

export default async function StudentProgressStrip({ classId, activities }: StudentProgressStripProps) {
  const { overall, byTerm } = await getStudentClassProgress(classId);
  if (overall.total === 0) return null;

  const pct = Math.round((overall.done / overall.total) * 100);
  const left = overall.total - overall.done;

  const list = activities ?? [];
  const todo = list.filter((a) => ['not_started', 'open', 'late_window'].includes(a.status));
  const missed = list.filter((a) => a.status === 'missing').length;
  const graded = list.filter((a) => a.status === 'graded_unreturned' || a.status === 'graded_returned').length;
  const upNext = [...todo]
    .sort((x, y) => new Date(x.dueAt).getTime() - new Date(y.dueAt).getTime())
    .slice(0, 3);

  const terms = byTerm.filter((t) => (MODULE_TERMS as readonly string[]).includes(t.term));

  return (
    <div className="space-y-4">
      <section className="space-y-5 rounded-2xl border border-gray-200 bg-white p-5 shadow-sm">
        <h2 className="text-[15px] font-bold text-gray-900">Your progress</h2>

        <div className="flex items-center gap-4">
          <CompletionRing pct={pct} />
          <div>
            <p className="text-sm font-semibold text-gray-900">
              {overall.done} of {overall.total} submitted
            </p>
            <p className="text-xs text-gray-500">
              {left === 0 ? 'All caught up. Nice work!' : `${left} left to do.`}
            </p>
          </div>
        </div>

        {activities && (
          <div className="grid grid-cols-3 gap-2">
            {[
              { value: todo.length, label: 'To do' },
              { value: missed, label: 'Missed' },
              { value: graded, label: 'Graded' },
            ].map((s) => (
              <div key={s.label} className="rounded-xl bg-gray-50 px-2 py-2.5 text-center">
                <p className={`text-xl font-extrabold ${s.label === 'Missed' && s.value > 0 ? 'text-red-600' : 'text-gray-900'}`}>
                  {s.value}
                </p>
                <p className="text-[11px] text-gray-500">{s.label}</p>
              </div>
            ))}
          </div>
        )}

        <div className="space-y-3 border-t border-gray-100 pt-4">
          <p className="text-[11px] font-bold uppercase tracking-wider text-gray-500">By term</p>
          {terms.map((t) => (
            <div key={t.term} className="space-y-1.5">
              <div className="flex items-center justify-between gap-2 text-sm">
                <span className={`font-semibold ${TERM_TEXT[t.term]}`}>{MODULE_TERM_LABELS[t.term]}</span>
                <span className={t.total === 0 ? 'text-xs text-gray-400' : 'text-xs text-gray-600'}>
                  {t.total === 0 ? 'No activities yet' : `${t.done} / ${t.total}`}

                  className={`h-full rounded-full ${TERM_BAR[t.term]}`}
                  style={{ width: `${t.total === 0 ? 0 : Math.round((t.done / t.total) * 100)}%` }}
                />
              </div>

          ))}
        </div>
      </section>

function shuffle<T>(arr: T[]): T[] {
  const a = [...arr];
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}

// Compare answers ignoring markdown markers, spacing and case.
function normalize(s: string): string {
  return s.replace(/[*`]/g, '').replace(/\s+/g, ' ').trim().toLowerCase();
}

type Option = { text: string; correct: boolean };

function buildOptions(card: Card, allCards: Card[]): Option[] {
  const seen = new Set<string>([normalize(card.back)]);
  const distractors: string[] = [];
  for (const c of shuffle(allCards)) {
    if (c.id === card.id) continue;
    const n = normalize(c.back);
    if (!n || seen.has(n)) continue;
    seen.add(n);
    distractors.push(c.back);
    if (distractors.length === 3) break;
  }
  return shuffle([
    { text: card.back, correct: true },
    ...distractors.map((text) => ({ text, correct: false })),
  ]);
}

const IGNORED_KEYS = new Set(['Shift', 'Control', 'Alt', 'Meta', 'Tab', 'CapsLock']);

function isTypingTarget(target: EventTarget | null): boolean {
  const el = target as HTMLElement | null;
  if (!el) return false;
  return el.tagName === 'INPUT' || el.tagName === 'TEXTAREA' || el.isContentEditable;
}

function LearnStudy({ deck }: FlashcardStudyModeProps) {

  const total = cards.length;

  const initialQueue = useMemo(() => cards.map((c) => c.id), [cards]);

  const [queue, setQueue] = useState<string[]>(initialQueue);
  const [results, setResults] = useState<CardResult[]>([]);
  const [answer, setAnswer] = useState<LearnAnswer | null>(null);

      e.preventDefault();

    setResults([]);

    <div className="rounded-xl border border-gray-200 bg-white p-4 shadow-sm">
      {/* Header */}
      <div className="mb-4 flex items-center justify-between gap-3">

      {/* Progress bar */}

        e.preventDefault();

          <Layers className="h-4 w-4 text-purple-600" />

      <div className="flex items-center justify-between gap-3">

          type="button"
          onClick={() => goTo(index - 1)}
          disabled={isFirst}
          className="inline-flex items-center gap-1.5 rounded-md border border-gray-200 bg-white px-4 py-2 text-sm font-medium text-gray-700 hover:bg-gray-50 disabled:cursor-not-allowed disabled:opacity-40"
        >
          <ChevronLeft className="h-4 w-4" />
          Previous
        </button>

        <span className="hidden text-xs text-gray-400 sm:inline">
          Space to flip · ← → to move
        </span>

        {isLast ? (
          <button
            type="button"
            onClick={() => goTo(0)}
            className="inline-flex items-center gap-1.5 rounded-md bg-purple-600 px-4 py-2 text-sm font-medium text-white hover:bg-purple-700"
          >
            <RotateCcw className="h-4 w-4" />
            Start over
          </button>
        ) : (
          <button
            type="button"
            onClick={() => goTo(index + 1)}
            className="inline-flex items-center gap-1.5 rounded-md bg-purple-600 px-4 py-2 text-sm font-medium text-white hover:bg-purple-700"
          >
            Next
            <ChevronRight className="h-4 w-4" />
          </button>
        )}
      </div>
    </div>

function StudyComplete({
  knew,
  total,
  label,
  onRestart,
}: {
  knew: number;
  total: number;
  label: string;
  onRestart: () => void;
}) {
  const pct = total === 0 ? 0 : Math.round((knew / total) * 100);

  let message: string;
  let messageColor: string;
  if (pct >= 90) {
    message = 'Excellent! You really know this material.';
    messageColor = 'text-green-700';
  } else if (pct >= 70) {
    message = 'Good work — a few more reviews and you have it.';
    messageColor = 'text-blue-700';
  } else if (pct >= 50) {
    message = "Keep practicing. You're building familiarity.";
    messageColor = 'text-amber-700';
  } else {
    message = 'Take another pass through the lesson, then try again.';
    messageColor = 'text-rose-700';
  }

  return (
    <div className="rounded-xl border border-gray-200 bg-white p-6 text-center shadow-sm">
      <Trophy className="mx-auto mb-3 h-10 w-10 text-amber-500" />
      <h2 className="text-lg font-semibold text-gray-900">Study session complete</h2>
      <p className={`mt-2 text-sm font-medium ${messageColor}`}>{message}</p>

      <div className="mx-auto mt-6 max-w-xs">
        <div className="mb-2 flex items-center justify-between text-xs">
          <span className="font-medium text-gray-700">{label}</span>
          <span className="font-bold text-purple-700">
            {knew} / {total} ({pct}%)
          </span>
        </div>
        <div className="h-2 w-full overflow-hidden rounded-full bg-gray-100">
          <div className="h-full bg-purple-500" style={{ width: `${pct}%` }} />
        </div>
      </div>

      <button
        type="button"
        onClick={onRestart}
        className="mt-6 inline-flex items-center gap-2 rounded-md bg-purple-600 px-4 py-2 text-sm font-medium text-white hover:bg-purple-700"
      >
        <RotateCcw className="h-4 w-4" />
        Study again
      </button>
    </div>
  );
}
